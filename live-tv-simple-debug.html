Live TV Debug Test

Manual Testing Instructions:

  1. Click "Open Live TV Page" below
  2. Wait for the page to load completely
  3. Open browser developer tools (F12)
  4. Copy and paste the debug script below into the console
  5. Watch the console output to see what happens

Debug Script (copy this to console):

// Live TV Debug Script console.log('🚀 Starting Live TV Debug...'); // Step 1: Find network cards setTimeout(() => { console.log('📺 Looking for network cards...'); const networkCards = document.querySelectorAll('[class*="cursor-pointer"], .cursor-pointer'); console.log('Found network cards:', networkCards.length); if (networkCards.length > 0) { console.log('🖱️ Clicking first network...'); networkCards[0].click(); // Step 2: Find Watch Live buttons after network click setTimeout(() => { console.log('🔍 Looking for Watch Live buttons...'); const watchButtons = Array.from(document.querySelectorAll('button')).filter(btn => btn.textContent && btn.textContent.includes('Watch Live') ); console.log('Found Watch Live buttons:', watchButtons.length); if (watchButtons.length > 0) { console.log('Button text:', watchButtons[0].textContent); console.log('Button disabled:', watchButtons[0].disabled); console.log('🎬 Clicking Watch Live button...'); watchButtons[0].click(); // Step 3: Check if video player modal opens setTimeout(() => { const modal = document.querySelector('[role="dialog"], .fixed.inset-0'); const video = document.querySelector('video'); console.log('Modal found:', !!modal); console.log('Video element found:', !!video); if (video) { console.log('Video src:', video.src); } if (modal || video) { console.log('✅ SUCCESS: Video player opened!'); } else { console.log('❌ FAILURE: Video player did not open'); } }, 3000); } else { console.log('❌ No Watch Live buttons found'); } }, 3000); } else { console.log('❌ No network cards found'); } }, 2000);
